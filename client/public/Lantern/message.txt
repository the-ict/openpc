// ─── Three.js usage example ──────────────────────────────────────────────
import * as THREE from 'three'
import { gsap } from 'gsap'
import looksData from './looks.json'

function applyLook(lookName, camera, lights) {
  const look = looksData.looks[lookName]

  // Camera
  const cam = look.cameras[0].threejs
  gsap.to(camera.position, { ...cam.position, duration: 1.2, ease: 'power2.inOut' })
  gsap.to(camera.rotation, {
    x: THREE.MathUtils.degToRad(cam.rotation_deg.x),
    y: THREE.MathUtils.degToRad(cam.rotation_deg.y),
    z: THREE.MathUtils.degToRad(cam.rotation_deg.z),
    duration: 1.2, ease: 'power2.inOut'
  })
  camera.fov = cam.fov_vertical_deg
  camera.updateProjectionMatrix()

  // Lights (match by name prefix)
  look.lights.forEach(lightData => {
    const threeLight = lights.find(l => l.name === lightData.name)
    if (!threeLight) return
    threeLight.color.set(lightData.threejs.color_hex)
    threeLight.intensity = lightData.threejs.intensity
    threeLight.position.set(
      lightData.threejs.position.x,
      lightData.threejs.position.y,
      lightData.threejs.position.z
    )
  })
}
